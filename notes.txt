sudo mv kubectl /usr/local/bin/
kubectl version --client

# 5. Install eksctl
curl -s --location "https://github.com/weaveworks/eksctl/releases/latest/download/eksctl_Linux_amd64.tar.gz" | tar xz
sudo mv eksctl /usr/local/bin/
eksctl version


eksctl create cluster \
  --name dev-cluster \
  --region ap-south-1 \
  --version 1.30 \
  --nodegroup-name standard-workers \
  --node-type t3.medium \
  --nodes 2 \
  --nodes-min 1 \
  --nodes-max 3 \
  --managed

-->connect kubectl to cluster
aws eks update-kubeconfig --region ap-south-1 --name dev-cluster
